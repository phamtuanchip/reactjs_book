TypeScript với React

TypeScript biến React từ ngôn ngữ dynamic thành statically typed — bắt lỗi trước khi chạy, autocomplete thông minh, refactor an toàn. Ngành tuyển dụng hiện nay gần như bắt buộc TypeScript.

21.3 Typing Props

// 1. Interface (khuyến nghị cho props)
interface ButtonProps {
  label: string;
  variant?: 'primary' | 'secondary' | 'danger'; // Optional union type
  size?: 'sm' | 'md' | 'lg';
  disabled?: boolean;
  onClick?: (event: React.MouseEvent<HTMLButtonElement>) => void;
  children?: React.ReactNode; // Bất kỳ JSX nào
  className?: string;
}

function Button({ label, variant = 'primary', size = 'md', onClick, disabled }: ButtonProps) {
  return (
    <button
      className={`btn btn-${variant} btn-${size}`}
      onClick={onClick}
      disabled={disabled}
    >
      {label}
    </button>
  );
}

// 2. Extending HTML element props
interface InputProps extends React.InputHTMLAttributes<HTMLInputElement> {
  label: string;
  error?: string;
}

function Input({ label, error, ...rest }: InputProps) {
  return (
    <div>
      <label>{label}</label>
      <input {...rest} />
      {error && <span className="error">{error}</span>}
    </div>
  );
}
// Giờ Input nhận tất cả props của <input> (type, placeholder, value...) + label + error

// 3. Generic component
interface ListProps<T> {
  items: T[];
  renderItem: (item: T, index: number) => React.ReactNode;
  keyExtractor: (item: T) => string;
  emptyMessage?: string;
}

function List<T>({ items, renderItem, keyExtractor, emptyMessage = 'Không có dữ liệu' }: ListProps<T>) {
  if (items.length === 0) return <p>{emptyMessage}</p>;
  return <ul>{items.map((item, i) => <li key={keyExtractor(item)}>{renderItem(item, i)}</li>)}</ul>;
}

// Dùng với bất kỳ type nào
<List
  items={users}          // User[]
  keyExtractor={u => u.id.toString()}
  renderItem={u => <span>{u.name}</span>}
/>

21.4 Typing Hooks

// useState — tự infer từ initial value
const [count, setCount] = useState(0);           // number
const [name, setName] = useState('');             // string
const [user, setUser] = useState<User | null>(null); // User | null (cần explicit)

// Khi có Union type phức tạp
type Status = 'idle' | 'loading' | 'success' | 'error';
const [status, setStatus] = useState<Status>('idle');

// useRef
const inputRef = useRef<HTMLInputElement>(null);
const timerRef = useRef<ReturnType<typeof setTimeout>>(null);
const countRef = useRef<number>(0);

// useReducer
type TodoAction =
  | { type: 'ADD'; payload: string }
  | { type: 'TOGGLE'; payload: number }
  | { type: 'DELETE'; payload: number };

interface Todo {
  id: number;
  text: string;
  done: boolean;
}

function todoReducer(state: Todo[], action: TodoAction): Todo[] {
  switch (action.type) {
    case 'ADD': return [...state, { id: Date.now(), text: action.payload, done: false }];
    case 'TOGGLE': return state.map(t => t.id === action.payload ? { ...t, done: !t.done } : t);
    case 'DELETE': return state.filter(t => t.id !== action.payload);
    default: return state; // TS biết action.type là never ở đây
  }
}

const [todos, dispatch] = useReducer(todoReducer, []);

// useContext
interface ThemeContextType {
  theme: 'light' | 'dark';
  toggleTheme: () => void;
}

const ThemeContext = createContext<ThemeContextType | null>(null);

function useTheme(): ThemeContextType {
  const ctx = useContext(ThemeContext);
  if (!ctx) throw new Error('useTheme phải được dùng trong ThemeProvider');
  return ctx; // Giờ ctx KHÔNG còn null
}

21.6 Utility Types hữu dụng

interface User {
  id: number;
  name: string;
  email: string;
  password: string;
  role: 'admin' | 'user';
  createdAt: Date;
}

// Partial — tất cả fields thành optional
type UserUpdate = Partial<User>;
// { id?: number; name?: string; email?: string; ... }

// Required — tất cả fields thành required
type RequiredUser = Required<User>;

// Pick — chọn một số fields
type UserPreview = Pick<User, 'id' | 'name' | 'role'>;
// { id: number; name: string; role: 'admin' | 'user' }

// Omit — bỏ bớt fields
type PublicUser = Omit<User, 'password'>;
// Không có password

// Record — map type
type RolePermissions = Record<User['role'], string[]>;
// { admin: string[]; user: string[] }

// ReturnType — type của giá trị trả về
function createUser(data: Partial<User>): User {
  return { id: 1, ...data } as User;
}
type NewUser = ReturnType<typeof createUser>; // User

// Readonly — không thể mutate
type ImmutableUser = Readonly<User>;

// Thực tế: API response types
type ApiResponse<T> = {
  data: T;
  status: number;
  message: string;
};

type UsersResponse = ApiResponse<User[]>;
// { data: User[]; status: number; message: string }

Câu hỏi ôn bài

1. interface và type trong TypeScript khác nhau thế nào?

  • A. Hoàn toàn giống nhau
  • B. interface có thể extend và được merge (declaration merging); type có thể dùng union/intersection và conditional types phức tạp hơn
  • C. type chỉ dùng cho primitives
  • D. interface chỉ dùng trong class

Đáp án: B — Convention: dùng interface cho props, objects, class contracts (có thể extend). Dùng type cho union types, computed types, và type aliases phức tạp. Trong React: Props dùng interface.

2. Discriminated union là gì?

  • A. Union của primitive types
  • B. Union type trong đó mỗi member có một literal field chung để phân biệt (như trường type trong action)
  • C. Union chỉ có 2 members
  • D. Union của class types

Đáp án: B — type Action = | { type: 'ADD'; payload: string } | { type: 'DELETE'; id: number } — type là discriminant. Trong switch/if (type === 'ADD'), TS biết payload có kiểu string. Pattern này rất phổ biến cho reducer actions.

3. strictNullChecks: true có tác dụng gì?

  • A. Cấm dùng null trong code
  • B. null và undefined phải được xử lý tường minh — không thể assign null vào string mà không check
  • C. Tự động throw error khi gặp null
  • D. Không có tác dụng thực tế

Đáp án: B — Không có strict: const name: string = null — OK. Với strict: Error! Phải dùng string | null. Buộc lập trình viên xử lý null case, tránh runtime error "Cannot read property of null".

4. React.ReactNode khác React.ReactElement ở điểm gì?

  • A. Giống nhau
  • B. ReactNode bao gồm string, number, null, boolean và ReactElement — dùng cho children. ReactElement chỉ là JSX element
  • C. ReactElement bao gồm nhiều hơn
  • D. ReactNode chỉ dùng trong class component

Đáp án: B — Props children nên type là React.ReactNode vì children có thể là JSX, string, number, null, undefined, fragment... ReactElement chỉ là result của JSX (React.createElement).

5. Tại sao nên dùng Omit<User, 'password'> thay vì tạo interface mới?

  • A. Omit nhanh hơn
  • B. Đảm bảo tính nhất quán — khi thêm field vào User, PublicUser tự động cập nhật (trừ password)
  • C. Interface mới không hoạt động
  • D. Không có lý do, chỉ là convention

Đáp án: B — Single source of truth: User là "nguồn sự thật". Thêm field avatar vào User → PublicUser tự có avatar. Nếu tạo interface riêng phải cập nhật 2 nơi → dễ mismatch. Utility types giữ code DRY.
