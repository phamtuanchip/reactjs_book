// prisma/schema.prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model User {
  id           String   @id @default(cuid())
  name         String
  email        String   @unique
  passwordHash String
  avatar       String?
  createdAt    DateTime @default(now())

  memberships  ProjectMember[]
  assignedTasks Task[] @relation("AssignedTasks")
  activities   Activity[]
}

model Project {
  id          String   @id @default(cuid())
  name        String
  description String?
  color       String   @default("#6366f1")
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  members     ProjectMember[]
  tasks       Task[]
}

model ProjectMember {
  id        String   @id @default(cuid())
  role      Role     @default(MEMBER)
  joinedAt  DateTime @default(now())

  user      User     @relation(fields: [userId], references: [id])
  userId    String
  project   Project  @relation(fields: [projectId], references: [id])
  projectId String

  @@unique([userId, projectId])
}

model Task {
  id          String     @id @default(cuid())
  title       String
  description String?
  status      TaskStatus @default(BACKLOG)
  priority    Priority   @default(MEDIUM)
  dueDate     DateTime?
  position    Float      @default(0)  // Cho drag-and-drop ordering
  createdAt   DateTime   @default(now())
  updatedAt   DateTime   @updatedAt

  project     Project    @relation(fields: [projectId], references: [id])
  projectId   String
  assignee    User?      @relation("AssignedTasks", fields: [assigneeId], references: [id])
  assigneeId  String?
  labels      Label[]
  activities  Activity[]
}

enum TaskStatus { BACKLOG TODO IN_PROGRESS DONE }
enum Priority  { LOW MEDIUM HIGH URGENT }
enum Role      { OWNER ADMIN MEMBER VIEWER }

25.4 Kanban Board với Drag and Drop

npm install @dnd-kit/core @dnd-kit/sortable @dnd-kit/utilities
'use client';
import {
  DndContext, DragEndEvent, DragOverlay,
  PointerSensor, useSensor, useSensors,
} from '@dnd-kit/core';
import { SortableContext, verticalListSortingStrategy } from '@dnd-kit/sortable';

const COLUMNS: TaskStatus[] = ['BACKLOG', 'TODO', 'IN_PROGRESS', 'DONE'];
const COLUMN_LABELS = {
  BACKLOG: 'Backlog',
  TODO: 'Cần làm',
  IN_PROGRESS: 'Đang làm',
  DONE: 'Hoàn thành',
};

function KanbanBoard({ projectId }: { projectId: string }) {
  const { data: tasks = [] } = useTasks(projectId);
  const { mutate: updateTask } = useUpdateTask();
  const [activeTask, setActiveTask] = useState<Task | null>(null);

  const sensors = useSensors(
    useSensor(PointerSensor, { activationConstraint: { distance: 8 } })
  );

  const tasksByStatus = COLUMNS.reduce((acc, status) => {
    acc[status] = tasks.filter(t => t.status === status)
                       .sort((a, b) => a.position - b.position);
    return acc;
  }, {} as Record<TaskStatus, Task[]>);

  const handleDragEnd = ({ active, over }: DragEndEvent) => {
    if (!over || active.id === over.id) return;

    const newStatus = over.data.current?.status as TaskStatus;
    const taskId = active.id as string;

    if (newStatus) {
      updateTask({ id: taskId, status: newStatus });
    }
  };

  return (
    <DndContext sensors={sensors} onDragEnd={handleDragEnd}
                onDragStart={({ active }) => setActiveTask(
                  tasks.find(t => t.id === active.id) || null
                )}>
      <div className="grid grid-cols-4 gap-4 h-full">
        {COLUMNS.map(status => (
          <KanbanColumn
            key={status}
            status={status}
            label={COLUMN_LABELS[status]}
            tasks={tasksByStatus[status]}
          />
        ))}
      </div>
      <DragOverlay>
        {activeTask && <TaskCard task={activeTask} isDragging />}
      </DragOverlay>
    </DndContext>
  );
}

25.5 Real-time với Server-Sent Events

// app/api/events/route.ts
export async function GET(request: Request) {
  const { searchParams } = new URL(request.url);
  const projectId = searchParams.get('projectId');

  const stream = new ReadableStream({
    start(controller) {
      const encoder = new TextEncoder();

      // Gửi heartbeat mỗi 30s để giữ connection
      const heartbeat = setInterval(() => {
        controller.enqueue(encoder.encode('data: heartbeat\n\n'));
      }, 30000);

      // Subscribe events từ event emitter hoặc Redis pub/sub
      const handler = (event: TaskEvent) => {
        if (event.projectId === projectId) {
          const data = JSON.stringify(event);
          controller.enqueue(encoder.encode(`data: ${data}\n\n`));
        }
      };

      eventEmitter.on('task-updated', handler);

      // Cleanup khi client disconnect
      request.signal.addEventListener('abort', () => {
        clearInterval(heartbeat);
        eventEmitter.off('task-updated', handler);
        controller.close();
      });
    }
  });

  return new Response(stream, {
    headers: {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache',
      'Connection': 'keep-alive',
    }
  });
}

// hooks/useSSE.ts
'use client';
export function useSSE(projectId: string) {
  const queryClient = useQueryClient();

  useEffect(() => {
    const source = new EventSource(`/api/events?projectId=${projectId}`);

    source.onmessage = (event) => {
      if (event.data === 'heartbeat') return;
      const data = JSON.parse(event.data);

      // Invalidate queries để refetch data mới
      queryClient.invalidateQueries({ queryKey: ['tasks', projectId] });
    };

    return () => source.close();
  }, [projectId, queryClient]);
}

Câu hỏi ôn bài — Tổng kết khóa học

1. Nhìn lại hành trình: từ HTML/CSS → JavaScript → React → TypeScript → Next.js. Tại sao học theo thứ tự này?

  • A. Thứ tự tùy ý, không quan trọng
  • B. Mỗi layer xây dựng trên layer trước — hiểu nền tảng giúp debug và hiểu tại sao tool hoạt động thế này
  • C. Chỉ vì truyền thống
  • D. Next.js có thể học trực tiếp không cần React

Đáp án: B — React là JavaScript. TypeScript là JavaScript + types. Next.js là React + server. Không hiểu nền tảng → không debug được, không biết tại sao component re-render, không hiểu async/await, không tự sửa lỗi TypeScript. Foundation matters.

2. Khi gặp bug khó, thứ tự debug nên như thế nào?

  • A. Google ngay → StackOverflow → copy code
  • B. Đọc error message kỹ → isolate vấn đề → console.log → React DevTools → tìm kiếm có chủ đích
  • C. Hỏi ChatGPT ngay
  • D. Restart server và hy vọng tự sửa

Đáp án: B — Error message thường chỉ ra chính xác vấn đề. Isolate: comment code để thu hẹp nguyên nhân. console.log xem giá trị thực. React DevTools xem state/props. Chỉ search khi đã hiểu vấn đề là gì.

3. Senior developer khác junior ở điểm nào quan trọng nhất?

  • A. Biết nhiều thư viện hơn
  • B. Biết khi nào KHÔNG dùng thư viện — hiểu trade-offs, ưu tiên đúng vấn đề, code đơn giản và maintain được
  • C. Gõ code nhanh hơn
  • D. Làm việc nhiều giờ hơn

Đáp án: B — Junior: thêm Redux vì thấy mọi người dùng. Senior: dùng useState khi đủ, chỉ thêm Redux khi scale thực sự cần. Senior hiểu: YAGNI (You Aren't Gonna Need It), đọc code nhiều hơn viết, đơn giản là đỉnh cao của tinh tế.

4. Sau khóa học, bước tiếp theo nên là gì?

  • A. Học thêm nhiều framework khác (Vue, Angular, Svelte)
  • B. Xây dựng dự án thực tế, đăng lên GitHub, contribute open-source, apply intern/job
  • C. Làm lại tất cả bài tập từ đầu
  • D. Học thêm 5 khóa nữa trước khi apply job

Đáp án: B — Lý thuyết không thay thế được kinh nghiệm thực tế. Build something real, kể cả nhỏ. GitHub portfolio quan trọng hơn certificate. Contribute OSS = học từ code thực. Apply job dù chưa "ready" — bạn sẽ không bao giờ 100% ready.

5. Nguồn học tập đáng tin cậy nhất để tiếp tục phát triển?

  • A. Tutorial YouTube đa số
  • B. Official docs (React, Next.js, TypeScript, MDN) + đọc source code của thư viện bạn dùng
  • C. Chỉ dùng ChatGPT
  • D. StackOverflow mọi câu hỏi

Đáp án: B — Official docs: luôn cập nhật, chính xác, có examples. React docs (react.dev) là excellent. Next.js docs cực kỳ tốt. TypeScript handbook toàn diện. Đọc source code: hiểu cách thư viện hoạt động, học pattern từ expert, tự tự tin debug issue.

Chúc mừng bạn đã hoàn thành khóa học!

Nhớ rằng: code nhiều hơn đọc, build nhiều hơn học, share code của bạn. Cộng đồng React Việt Nam đang chờ bạn đóng góp.
