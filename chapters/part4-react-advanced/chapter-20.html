Testing trong React

Testing là kỹ năng phân biệt developer junior và senior. Chương này hướng dẫn viết tests cho React với Vitest và React Testing Library — test theo cách người dùng thực sự tương tác.

20.3 Unit Testing — Hàm thuần

// src/utils/validation.js
export function validateEmail(email) {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
}

export function formatCurrency(amount, currency = 'VND') {
  return new Intl.NumberFormat('vi-VN', { style: 'currency', currency }).format(amount);
}

// src/utils/validation.test.js
import { describe, it, expect } from 'vitest';
import { validateEmail, formatCurrency } from './validation';

describe('validateEmail', () => {
  it('trả về true với email hợp lệ', () => {
    expect(validateEmail('user@example.com')).toBe(true);
    expect(validateEmail('test.name+tag@domain.co')).toBe(true);
  });

  it('trả về false với email không hợp lệ', () => {
    expect(validateEmail('not-an-email')).toBe(false);
    expect(validateEmail('@nodomain.com')).toBe(false);
    expect(validateEmail('')).toBe(false);
    expect(validateEmail('space @email.com')).toBe(false);
  });
});

describe('formatCurrency', () => {
  it('định dạng tiền VND đúng', () => {
    expect(formatCurrency(1000000)).toContain('1.000.000');
  });

  it('dùng currency mặc định là VND', () => {
    expect(formatCurrency(50000)).toContain('₫');
  });
});

20.4 React Testing Library — Test component

Triết lý RTL: "Test như cách người dùng sử dụng" — không test implementation details.

import { render, screen } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import { describe, it, expect, vi } from 'vitest';

// Component cần test
function Counter({ initialCount = 0 }) {
  const [count, setCount] = useState(initialCount);
  return (
    <div>
      <p>Đếm: {count}</p>
      <button onClick={() => setCount(c => c + 1)}>Tăng</button>
      <button onClick={() => setCount(c => c - 1)}>Giảm</button>
      <button onClick={() => setCount(0)}>Reset</button>
    </div>
  );
}

describe('Counter', () => {
  it('hiển thị giá trị ban đầu', () => {
    render(<Counter initialCount={5} />);
    expect(screen.getByText('Đếm: 5')).toBeInTheDocument();
  });

  it('tăng khi click nút Tăng', async () => {
    const user = userEvent.setup();
    render(<Counter />);

    await user.click(screen.getByRole('button', { name: 'Tăng' }));
    expect(screen.getByText('Đếm: 1')).toBeInTheDocument();
  });

  it('reset về 0 khi click Reset', async () => {
    const user = userEvent.setup();
    render(<Counter initialCount={10} />);

    await user.click(screen.getByRole('button', { name: 'Reset' }));
    expect(screen.getByText('Đếm: 0')).toBeInTheDocument();
  });
});

20.5 Testing với Async và API Mock

import { render, screen, waitFor } from '@testing-library/react';
import { vi } from 'vitest';

// Mock global fetch
vi.stubGlobal('fetch', vi.fn());

describe('UserProfile', () => {
  it('hiển thị loading khi đang fetch', () => {
    fetch.mockImplementation(() => new Promise(() => {})); // Never resolve

    render(<UserProfile userId={1} />);
    expect(screen.getByText('Đang tải...')).toBeInTheDocument();
  });

  it('hiển thị user data khi fetch thành công', async () => {
    fetch.mockResolvedValue({
      ok: true,
      json: async () => ({ id: 1, name: 'Nguyễn An', email: 'an@example.com' })
    });

    render(<UserProfile userId={1} />);

    // Chờ async render
    await waitFor(() => {
      expect(screen.getByText('Nguyễn An')).toBeInTheDocument();
    });
    expect(screen.getByText('an@example.com')).toBeInTheDocument();
  });

  it('hiển thị error khi fetch thất bại', async () => {
    fetch.mockResolvedValue({ ok: false, status: 404 });

    render(<UserProfile userId={999} />);

    await waitFor(() => {
      expect(screen.getByText(/không tìm thấy/i)).toBeInTheDocument();
    });
  });

  afterEach(() => {
    vi.clearAllMocks(); // Reset mocks sau mỗi test
  });
});

20.6 Testing Form

describe('LoginForm', () => {
  it('hiển thị lỗi khi submit form rỗng', async () => {
    const user = userEvent.setup();
    render(<LoginForm />);

    await user.click(screen.getByRole('button', { name: 'Đăng nhập' }));

    expect(screen.getByText('Email là bắt buộc')).toBeInTheDocument();
    expect(screen.getByText('Mật khẩu là bắt buộc')).toBeInTheDocument();
  });

  it('gọi onSubmit với đúng data khi form hợp lệ', async () => {
    const user = userEvent.setup();
    const mockOnSubmit = vi.fn();
    render(<LoginForm onSubmit={mockOnSubmit} />);

    await user.type(screen.getByLabelText('Email'), 'test@example.com');
    await user.type(screen.getByLabelText('Mật khẩu'), 'password123');
    await user.click(screen.getByRole('button', { name: 'Đăng nhập' }));

    expect(mockOnSubmit).toHaveBeenCalledWith({
      email: 'test@example.com',
      password: 'password123'
    });
  });

  it('hiển thị lỗi email không hợp lệ', async () => {
    const user = userEvent.setup();
    render(<LoginForm />);

    await user.type(screen.getByLabelText('Email'), 'not-an-email');
    await user.tab(); // Trigger blur validation

    expect(screen.getByText(/email không hợp lệ/i)).toBeInTheDocument();
  });
});

20.7 Testing Queries — Tìm elements

// Accessible (recommended):

screen.getByLabelText('Email');
screen.getByPlaceholderText('Nhập email...');
screen.getByText('Xin chào');

// Semantic:
screen.getByAltText('Logo');
screen.getByTitle('Close dialog');

// Test ID (last resort):
screen.getByTestId('submit-btn'); // Thêm data-testid="submit-btn"

getBy...     // Fail nếu không tìm thấy

findBy...    // Async, chờ element xuất hiện

Bài tập 20.1: Test Todo List

// Viết test cho TodoApp từ Chương 11, bao gồm:
// 1. Render danh sách todos ban đầu
// 2. Thêm todo mới
// 3. Toggle done/undone
// 4. Xóa todo
// 5. Đếm todos chưa hoàn thành hiển thị đúng
// 6. Filter (All/Active/Completed) hoạt động đúng
// 7. "Xóa tất cả đã hoàn thành" hoạt động đúng
// 8. Input được clear sau khi thêm todo

// Bắt đầu với test này:
describe('TodoApp', () => {
  it('render empty state với message thích hợp', () => {
    render(<TodoApp />);
    expect(screen.getByText(/chưa có todo nào/i)).toBeInTheDocument();
  });

  it('thêm todo mới khi submit form', async () => {
    const user = userEvent.setup();
    render(<TodoApp />);

    await user.type(screen.getByPlaceholderText('Thêm todo...'), 'Học React');
    await user.keyboard('{Enter}');

    expect(screen.getByText('Học React')).toBeInTheDocument();
    expect(screen.getByPlaceholderText('Thêm todo...')).toHaveValue('');
  });
});

Câu hỏi ôn bài

1. Triết lý của React Testing Library là gì?

  • A. Test mọi implementation detail để đảm bảo code đúng
  • B. Test theo cách người dùng thực sự tương tác với UI, không phải cách component được implement
  • C. Chỉ test pure functions, không test components
  • D. Test 100% coverage mọi dòng code

Đáp án: B — Guiding principle của RTL: "The more your tests resemble the way your software is used, the more confidence they can give you." Test cái người dùng thấy và làm, không test internal state hay methods.

2. Khi nào dùng queryBy thay vì getBy?

  • A. Khi element có thể null
  • B. Khi muốn assert element KHÔNG tồn tại trong DOM
  • C. queryBy luôn tốt hơn getBy
  • D. Khi element là button

Đáp án: B — getBy throw error nếu không tìm thấy. queryBy return null. Dùng: expect(screen.queryByText('Lỗi')).not.toBeInTheDocument() để test rằng error message KHÔNG hiển thị.

3. userEvent khác fireEvent ở điểm gì?

  • A. userEvent chỉ test mobile, fireEvent test desktop
  • B. userEvent mô phỏng tương tác người dùng thực (click + hover + focus...), fireEvent chỉ trigger event đơn giản
  • C. fireEvent là async, userEvent là sync
  • D. Không có sự khác biệt

Đáp án: B — userEvent.click() = hover → focus → mousedown → mouseup → click (như người dùng thực). fireEvent.click() chỉ dispatch click event. userEvent tạo tests thực tế hơn.

4. Tại sao nên mock fetch trong unit/integration tests?

  • A. Fetch quá chậm
  • B. Tests không nên phụ thuộc vào network/server thực — cần deterministic, nhanh, không phụ thuộc external service
  • C. Fetch không hoạt động trong test environment
  • D. Để tiết kiệm API calls

Đáp án: B — Tests phải: (1) Nhanh, (2) Ổn định (không fail ngẫu nhiên do network), (3) Kiểm soát được response. Mock cho phép test cả happy path và error cases một cách nhất quán.

5. Query nào được khuyến nghị nhất theo thứ tự ưu tiên của RTL?

  • A. getByTestId
  • B. getByClassName
  • C. getByRole
  • D. getByText

Đáp án: C — getByRole tốt nhất vì: (1) Phản ánh cách assistive technology truy cập UI, (2) Test accessibility, (3) Ổn định — không bị break khi thay đổi CSS hay text. getByTestId là last resort khi không có cách khác.
