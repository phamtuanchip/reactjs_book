Form Nâng cao

Form phức tạp với nhiều trường, validation, và multi-step là thách thức lớn trong React. React Hook Form kết hợp Zod giải quyết vấn đề này một cách elegant, performant và type-safe.

18.2 React Hook Form — Hiệu năng cao

npm install react-hook-form
import { useForm } from 'react-hook-form';

function LoginForm() {
  const {
    register,          // Đăng ký input với form
    handleSubmit,      // Wrapper cho submit handler
    formState: { errors, isSubmitting, isValid }, // Trạng thái form
    watch,             // Theo dõi giá trị field realtime
    reset,             // Reset form về giá trị ban đầu
    setValue,          // Set giá trị field bằng code
    getValues,         // Đọc giá trị hiện tại
  } = useForm({
    defaultValues: {
      email: '',
      password: '',
      rememberMe: false,
    },
    mode: 'onBlur',  // Validate khi rời trường (onBlur/onChange/onSubmit)
  });

  const onSubmit = async (data) => {
    // data có đầy đủ giá trị form, đã qua validation
    console.log(data); // { email: "...", password: "..." }
    await login(data);
  };

  return (
    <form onSubmit={handleSubmit(onSubmit)}>
      <div>
        <label>Email</label>
        <input
          {...register('email', {
            required: 'Email là bắt buộc',
            pattern: {
              value: /^[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}$/i,
              message: 'Email không hợp lệ'
            }
          })}
        />
        {errors.email && <span className="error">{errors.email.message}</span>}
      </div>

      <div>
        <label>Mật khẩu</label>
        <input
          type="password"
          {...register('password', {
            required: 'Mật khẩu là bắt buộc',
            minLength: { value: 8, message: 'Tối thiểu 8 ký tự' }
          })}
        />
        {errors.password && <span className="error">{errors.password.message}</span>}
      </div>

      <input type="checkbox" {...register('rememberMe')} />

      <button type="submit" disabled={isSubmitting}>
        {isSubmitting ? 'Đang đăng nhập...' : 'Đăng nhập'}
      </button>
    </form>
  );
}

18.3 Validation với Zod

npm install zod @hookform/resolvers
import { z } from 'zod';
import { zodResolver } from '@hookform/resolvers/zod';

// Định nghĩa schema validation
const registerSchema = z.object({
  name: z.string()
    .min(2, 'Tên phải có ít nhất 2 ký tự')
    .max(50, 'Tên quá dài'),

  email: z.string()
    .email('Email không hợp lệ'),

  password: z.string()
    .min(8, 'Mật khẩu tối thiểu 8 ký tự')
    .regex(/[A-Z]/, 'Cần ít nhất 1 chữ hoa')
    .regex(/[0-9]/, 'Cần ít nhất 1 số'),

  confirmPassword: z.string(),

  birthDate: z.string()
    .refine((date) => {
      const age = new Date().getFullYear() - new Date(date).getFullYear();
      return age >= 18;
    }, 'Phải đủ 18 tuổi'),

  role: z.enum(['user', 'admin', 'moderator']),

  agreeToTerms: z.literal(true, {
    errorMap: () => ({ message: 'Bạn phải đồng ý điều khoản' })
  }),
}).refine(data => data.password === data.confirmPassword, {
  message: 'Mật khẩu không khớp',
  path: ['confirmPassword'],
});

// TypeScript type tự động từ schema
type RegisterFormData = z.infer<typeof registerSchema>;

// Dùng với React Hook Form
function RegisterForm() {
  const { register, handleSubmit, formState: { errors } } = useForm({
    resolver: zodResolver(registerSchema),
  });

  // TypeScript biết chính xác type của data!
  const onSubmit = (data: RegisterFormData) => {
    console.log(data);
  };

  return <form onSubmit={handleSubmit(onSubmit)}>...</form>;
}

18.4 Multi-step Form

// Multi-step form với React Hook Form
const STEPS = ['Thông tin cá nhân', 'Tài khoản', 'Xác nhận'];

const step1Schema = z.object({
  firstName: z.string().min(2),
  lastName: z.string().min(2),
  phone: z.string().regex(/^0\d{9}$/),
});

const step2Schema = z.object({
  email: z.string().email(),
  password: z.string().min(8),
  confirmPassword: z.string(),
}).refine(data => data.password === data.confirmPassword, {
  message: 'Mật khẩu không khớp',
  path: ['confirmPassword']
});

const schemas = [step1Schema, step2Schema];

function MultiStepForm() {
  const [step, setStep] = useState(0);
  const [formData, setFormData] = useState({});

  const { register, handleSubmit, formState: { errors } } = useForm({
    resolver: zodResolver(schemas[step]),
  });

  const handleNext = handleSubmit((data) => {
    setFormData(prev => ({ ...prev, ...data }));
    if (step < STEPS.length - 1) {
      setStep(s => s + 1);
    } else {
      // Submit cuối cùng
      submitForm({ ...formData, ...data });
    }
  });

  return (
    <div>
      {/* Progress indicator */}
      <div className="steps">
        {STEPS.map((s, i) => (
          <div key={i} className={`step ${i <= step ? 'active' : ''}`}>
            {s}
          </div>
        ))}
      </div>

      <form onSubmit={handleNext}>
        {step === 0 && (
          <>
            <input {...register('firstName')} placeholder="Họ" />
            {errors.firstName && <span>{errors.firstName.message}</span>}
            <input {...register('lastName')} placeholder="Tên" />
            <input {...register('phone')} placeholder="Số điện thoại" />
          </>
        )}

        {step === 1 && (
          <>
            <input {...register('email')} type="email" />
            <input {...register('password')} type="password" />
            <input {...register('confirmPassword')} type="password" />
          </>
        )}

        {step === 2 && (
          <div>
            <h3>Xác nhận thông tin</h3>
            <pre>{JSON.stringify(formData, null, 2)}</pre>
          </div>
        )}

        <div>
          {step > 0 && (
            <button type="button" onClick={() => setStep(s => s - 1)}>
              Quay lại
            </button>
          )}
          <button type="submit">
            {step === STEPS.length - 1 ? 'Hoàn tất' : 'Tiếp theo'}
          </button>
        </div>
      </form>
    </div>
  );
}

Bài tập 18.1: Form đăng ký hoàn chỉnh với validation

// Xây dựng form đăng ký với:
// Schema Zod:
// - username: 3-20 ký tự, chỉ chữ/số/gạch dưới

// - bio: max 200 ký tự (optional)
// - website: URL hợp lệ hoặc rỗng (optional)
// - newsletter: boolean
//
// Features:
// - Hiển thị error ngay khi rời trường (onBlur)
// - Password strength indicator
// - Character count cho bio
// - Submit button disabled khi form invalid
// - Toast notification sau khi submit thành công

Câu hỏi ôn bài

1. Tại sao React Hook Form hiệu năng hơn controlled components thông thường?

  • A. Nó chạy trên một thread riêng
  • B. Sử dụng uncontrolled inputs với refs — không re-render component khi user gõ
  • C. Tự động tối ưu code
  • D. Sử dụng Web Workers

Đáp án: B — RHF dùng refs để đọc giá trị (uncontrolled), chỉ re-render khi validation state thay đổi (errors, isValid). Controlled components re-render sau mỗi keystroke, gây lag với form nhiều trường.

2. mode: 'onBlur' trong React Hook Form có nghĩa là?

  • A. Validate khi form bị blur (mất focus)
  • B. Validate khi user rời khỏi từng trường (blur event)
  • C. Validate chỉ khi submit
  • D. Không validate

Đáp án: B — 'onBlur': validate khi user rời trường — UX tốt nhất cho hầu hết form. 'onChange': validate realtime khi gõ. 'onSubmit': chỉ validate khi submit.

3. Zod khác với việc validate thủ công như thế nào?

  • A. Zod chậm hơn nhưng chính xác hơn
  • B. Zod cung cấp schema declaration, tự động generate TypeScript types, error messages nhất quán, và composable
  • C. Zod chỉ hoạt động trên server
  • D. Zod chỉ validate string

Đáp án: B — Zod: khai báo schema một lần, dùng cho cả validation và TypeScript types (z.infer<typeof schema>). Có thể reuse schema giữa frontend và backend. Error messages nhất quán và có thể customize.

4. Trong multi-step form, tại sao không nên reset toàn bộ form khi chuyển step?

  • A. Reset gây lỗi
  • B. Dữ liệu từ step trước cần được giữ lại để submit ở step cuối
  • C. RHF không hỗ trợ reset
  • D. Reset quá chậm

Đáp án: B — Mỗi step thu thập một phần dữ liệu. Khi hoàn thành step, merge data vào state chung (setFormData(prev => ({ ...prev, ...stepData }))). Step cuối submit toàn bộ formData.

5. z.refine() trong Zod dùng để làm gì?

  • A. Định dạng lại dữ liệu
  • B. Thêm validation custom không có sẵn trong Zod (ví dụ: kiểm tra password khớp nhau, tuổi đủ 18)
  • C. Chuyển đổi kiểu dữ liệu
  • D. Tối ưu validation

Đáp án: B — .refine(fn, message) nhận validator function tùy chỉnh, phù hợp cho cross-field validation (password === confirmPassword) hoặc validation phức tạp (gọi API kiểm tra username tồn tại).
