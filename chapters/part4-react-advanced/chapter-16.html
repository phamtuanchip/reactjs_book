16.2 Context API (Built-in React)

// Tổ chức Context tốt hơn — kết hợp với useReducer
import { createContext, useContext, useReducer } from 'react';

// State
const initialState = {
  user: null,
  isAuthenticated: false,
  theme: 'dark',
};

// Reducer
function appReducer(state, action) {
  switch (action.type) {
    case 'LOGIN':
      return { ...state, user: action.payload, isAuthenticated: true };
    case 'LOGOUT':
      return { ...state, user: null, isAuthenticated: false };
    case 'SET_THEME':
      return { ...state, theme: action.payload };
    default:
      return state;
  }
}

// Context
const AppContext = createContext(null);

// Provider
export function AppProvider({ children }) {
  const [state, dispatch] = useReducer(appReducer, initialState);

  return (
    <AppContext.Provider value={{ state, dispatch }}>
      {children}
    </AppContext.Provider>
  );
}

// Custom hook
export function useApp() {
  const context = useContext(AppContext);
  if (!context) throw new Error('useApp must be used within AppProvider');
  return context;
}

// Dùng trong component
function UserAvatar() {
  const { state, dispatch } = useApp();
  return (
    <div>
      <img src={state.user?.avatar} />
      <button onClick={() => dispatch({ type: 'LOGOUT' })}>Logout</button>
    </div>
  );
}
⚠️ Hiệu năng Context

Khi context value thay đổi, TẤT CẢ component dùng useContext đó đều re-render. Tách context theo chức năng (UserContext, ThemeContext, CartContext riêng biệt) để giảm re-render không cần thiết.

16.3 Zustand — State Management nhẹ và đơn giản

# Cài đặt
npm install zustand
import { create } from 'zustand';

// Tạo store
const useCartStore = create((set, get) => ({
  // State
  items: [],
  total: 0,

  // Actions
  addItem: (product) => set((state) => {
    const existing = state.items.find(i => i.id === product.id);
    if (existing) {
      return {
        items: state.items.map(i =>
          i.id === product.id ? { ...i, qty: i.qty + 1 } : i
        )
      };
    }
    return { items: [...state.items, { ...product, qty: 1 }] };
  }),

  removeItem: (id) => set((state) => ({
    items: state.items.filter(i => i.id !== id)
  })),

  clearCart: () => set({ items: [], total: 0 }),

  // Computed (dùng get())
  getTotalPrice: () => {
    const { items } = get();
    return items.reduce((sum, item) => sum + item.price * item.qty, 0);
  }
}));

// Dùng trong component — CỰC ĐƠN GIẢN!
function CartButton() {
  const { items, addItem } = useCartStore();
  return <span>{items.length}</span>;
}

function ProductCard({ product }) {
  const addItem = useCartStore(state => state.addItem); // Chỉ subscribe addItem
  return <button onClick={() => addItem(product)}>Thêm vào giỏ</button>;
}

// Zustand với persist (lưu vào localStorage)
import { persist } from 'zustand/middleware';

const useThemeStore = create(
  persist(
    (set) => ({
      theme: 'dark',
      setTheme: (theme) => set({ theme }),
    }),
    { name: 'theme-storage' } // localStorage key
  )
);

16.4 Redux Toolkit

# Cài đặt
npm install @reduxjs/toolkit react-redux
// store/cartSlice.js — Slice = reducer + actions
import { createSlice } from '@reduxjs/toolkit';

const cartSlice = createSlice({
  name: 'cart',
  initialState: { items: [] },
  reducers: {
    addItem: (state, action) => {
      const existing = state.items.find(i => i.id === action.payload.id);
      if (existing) {
        existing.qty += 1;
      } else {
        state.items.push({ ...action.payload, qty: 1 });
      }
      // Redux Toolkit dùng Immer → có thể "mutate" trực tiếp!
    },
    removeItem: (state, action) => {
      state.items = state.items.filter(i => i.id !== action.payload);
    },
    clearCart: (state) => {
      state.items = [];
    }
  }
});

export const { addItem, removeItem, clearCart } = cartSlice.actions;
export default cartSlice.reducer;

// store/index.js
import { configureStore } from '@reduxjs/toolkit';
import cartReducer from './cartSlice';

export const store = configureStore({
  reducer: {
    cart: cartReducer,
  }
});

// main.jsx
import { Provider } from 'react-redux';
<Provider store={store}><App /></Provider>

// Dùng trong component
import { useSelector, useDispatch } from 'react-redux';

function CartButton() {
  const items = useSelector(state => state.cart.items);
  const dispatch = useDispatch();

  return (
    <div>
      <span>{items.length}</span>
      <button onClick={() => dispatch(clearCart())}>Xóa giỏ</button>
    </div>
  );
}

Câu hỏi ôn bài

1. Tại sao Context API có thể gây vấn đề hiệu năng nếu dùng sai?

  • A. Context chậm hơn useState về tốc độ xử lý
  • B. Khi context value thay đổi, TẤT CẢ consumer đều re-render, kể cả những component không dùng phần thay đổi
  • C. Context không hỗ trợ TypeScript
  • D. Context chỉ hoạt động với class components

Đáp án: B — Giải pháp: tách context nhỏ theo chức năng (không để tất cả trong 1 context lớn), hoặc dùng Zustand/Redux vì chúng chỉ re-render component subscribe đúng slice state đó.

2. Điểm khác biệt lớn nhất giữa Zustand và Redux Toolkit?

  • A. Zustand là TypeScript, Redux là JavaScript
  • B. Zustand đơn giản hơn, ít boilerplate hơn; Redux Toolkit có nhiều tính năng hơn (DevTools, middleware, RTK Query)
  • C. Redux nhanh hơn Zustand
  • D. Chúng hoàn toàn giống nhau

Đáp án: B — Zustand: ~1KB, API đơn giản, không cần Provider. Redux Toolkit: mạnh mẽ hơn, có Redux DevTools, RTK Query cho data fetching, nhưng phức tạp hơn và cần setup nhiều hơn.

3. Trong Redux Toolkit, tại sao có thể "mutate" state trực tiếp trong reducer?

  • A. Redux Toolkit thay đổi quy tắc immutability
  • B. Redux Toolkit dùng thư viện Immer bên dưới, tự convert mutations thành immutable updates
  • C. Redux Toolkit chạy trên server nên mutation là OK
  • D. Đây là tính năng của JavaScript mới nhất

Đáp án: B — Immer cho phép viết code "mutation style" nhưng bên dưới tạo bản copy immutable. Điều này giúp code dễ đọc hơn so với spread operator liên tục trong Redux thuần.

4. Selector trong Zustand (useStore(state => state.cart.items)) có lợi ích gì?

  • A. Làm code chạy nhanh hơn
  • B. Component chỉ re-render khi phần state đó thay đổi, không re-render khi phần khác thay đổi
  • C. Tự động cache dữ liệu
  • D. Không có lợi ích đặc biệt

Đáp án: B — Selector tối ưu: component chỉ subscribe phần state nó cần. Nếu subscribe toàn bộ store (useStore()), component re-render khi BẤT KỲ phần nào của store thay đổi.

5. "Server state" và "client state" khác nhau thế nào?

  • A. Server state lưu trên server, client state lưu trong localStorage
  • B. Server state: dữ liệu từ API (cache, loading, error); Client state: UI state (modal open, theme, filter). Zustand/Redux cho client state, TanStack Query cho server state
  • C. Không có sự khác biệt, dùng Redux cho cả hai
  • D. Server state là real-time, client state là static

Đáp án: B — Phân biệt quan trọng: server state (data từ API) cần caching, background refetch, invalidation → dùng TanStack Query. Client state (UI preferences, form data) → dùng useState/Zustand.
