Tương tác với API và Data Fetching

TanStack Query (trước đây là React Query) giải quyết vấn đề "server state" — caching, loading, error, background refresh, pagination. Đây là công cụ biến đổi cách bạn tương tác với API trong React.

17.1 Vấn đề với fetch thuần trong useEffect

// Cách làm "naive" — có nhiều vấn đề:
function PostList() {
  const [posts, setPosts] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    fetch('/api/posts')
      .then(r => r.json())
      .then(setPosts)
      .catch(setError)
      .finally(() => setLoading(false));
  }, []);

  // Vấn đề:
  // ❌ Không cache — mỗi lần navigate đến trang này fetch lại từ đầu
  // ❌ Không background refresh — data cũ khi user quay lại tab
  // ❌ Không retry khi lỗi
  // ❌ Không deduplicate — nhiều component fetch cùng URL = nhiều request
  // ❌ Race conditions
  // ❌ Boilerplate loading/error lặp đi lặp lại
}

17.2 TanStack Query — Setup

npm install @tanstack/react-query @tanstack/react-query-devtools
// main.jsx — Setup QueryClient
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
import { ReactQueryDevtools } from '@tanstack/react-query-devtools';

const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      staleTime: 5 * 60 * 1000,     // 5 phút trước khi data "cũ"
      cacheTime: 10 * 60 * 1000,    // 10 phút giữ trong cache
      retry: 3,                      // Retry 3 lần khi lỗi
      refetchOnWindowFocus: true,    // Refetch khi user quay lại tab
    }
  }
});

ReactDOM.createRoot(document.getElementById('root')).render(
  <QueryClientProvider client={queryClient}>
    <App />
    <ReactQueryDevtools initialIsOpen={false} /> {/* DevTools */}
  </QueryClientProvider>
);

17.3 useQuery — Đọc dữ liệu

import { useQuery } from '@tanstack/react-query';

// API function (tách ra file riêng)
const fetchPosts = async () => {
  const response = await fetch('/api/posts');
  if (!response.ok) throw new Error('Lỗi tải bài viết');
  return response.json();
};

function PostList() {
  const {
    data: posts,      // Dữ liệu trả về
    isLoading,        // Lần đầu load, chưa có data
    isFetching,       // Đang fetch (kể cả background)
    isError,          // Có lỗi
    error,            // Đối tượng lỗi
    refetch,          // Hàm trigger fetch lại
    status,           // 'loading' | 'error' | 'success'
  } = useQuery({
    queryKey: ['posts'],           // Key để cache và deduplicate
    queryFn: fetchPosts,           // Hàm fetch
    staleTime: 30 * 1000,         // Override default
  });

  if (isLoading) return <Spinner />;
  if (isError) return <Error message={error.message} onRetry={refetch} />;

  return (
    <ul>
      {posts.map(post => (
        <li key={post.id}>{post.title}</li>
      ))}
    </ul>
  );
}

// Query với parameters — query key thay đổi khi params thay đổi
function PostDetail({ postId }) {
  const { data: post } = useQuery({
    queryKey: ['post', postId],         // postId là dependency
    queryFn: () => fetchPost(postId),   // Fetch lại khi postId thay đổi
    enabled: !!postId,                  // Chỉ fetch khi postId có giá trị
  });
}

17.4 useMutation — Tạo/Cập nhật/Xóa dữ liệu

import { useMutation, useQueryClient } from '@tanstack/react-query';

function CreatePost() {
  const queryClient = useQueryClient();

  const createMutation = useMutation({
    mutationFn: (newPost) => fetch('/api/posts', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(newPost)
    }).then(r => r.json()),

    onSuccess: () => {
      // Invalidate cache để tự động refetch danh sách
      queryClient.invalidateQueries({ queryKey: ['posts'] });
    },

    onError: (error) => {
      console.error('Lỗi tạo bài:', error);
    }
  });

  const handleSubmit = (e) => {
    e.preventDefault();
    createMutation.mutate({ title, content });
  };

  return (
    <form onSubmit={handleSubmit}>
      {/* form fields */}
      <button disabled={createMutation.isPending}>
        {createMutation.isPending ? 'Đang tạo...' : 'Tạo bài'}
      </button>
      {createMutation.isError && <p className="error">{createMutation.error.message}</p>}
    </form>
  );
}

17.5 Optimistic Updates

// Cập nhật UI ngay lập tức, rollback nếu thất bại
const toggleLikeMutation = useMutation({
  mutationFn: (postId) => likePost(postId),

  onMutate: async (postId) => {
    // Hủy bất kỳ refetch nào đang chạy (tránh conflict)
    await queryClient.cancelQueries({ queryKey: ['post', postId] });

    // Lưu giá trị hiện tại để rollback nếu cần
    const previousPost = queryClient.getQueryData(['post', postId]);

    // Optimistic update — cập nhật cache ngay lập tức
    queryClient.setQueryData(['post', postId], old => ({
      ...old,
      liked: !old.liked,
      likesCount: old.liked ? old.likesCount - 1 : old.likesCount + 1
    }));

    return { previousPost }; // Context để dùng trong onError
  },

  onError: (err, postId, context) => {
    // Rollback về giá trị cũ khi lỗi
    queryClient.setQueryData(['post', postId], context.previousPost);
  },

  onSettled: (data, error, postId) => {
    // Sync lại với server dù thành công hay thất bại
    queryClient.invalidateQueries({ queryKey: ['post', postId] });
  }
});

17.6 Pagination và Infinite Scroll

// Pagination thông thường
function PaginatedList() {
  const [page, setPage] = useState(1);

  const { data } = useQuery({
    queryKey: ['posts', page],
    queryFn: () => fetchPosts(page),
    placeholderData: keepPreviousData, // Giữ data cũ khi đang tải trang mới
  });

  return (
    <div>
      {data?.posts.map(p => <PostCard key={p.id} post={p} />)}
      <button onClick={() => setPage(p => p - 1)} disabled={page === 1}>Trước</button>
      <span>Trang {page} / {data?.totalPages}</span>
      <button onClick={() => setPage(p => p + 1)} disabled={page === data?.totalPages}>Sau</button>
    </div>
  );
}

// Infinite scroll
import { useInfiniteQuery } from '@tanstack/react-query';

function InfinitePostList() {
  const {
    data,
    fetchNextPage,
    hasNextPage,
    isFetchingNextPage
  } = useInfiniteQuery({
    queryKey: ['posts', 'infinite'],
    queryFn: ({ pageParam = 1 }) => fetchPosts(pageParam),
    getNextPageParam: (lastPage) => lastPage.nextPage ?? undefined,
  });

  return (
    <div>
      {data?.pages.flatMap(page => page.posts).map(post => (
        <PostCard key={post.id} post={post} />
      ))}
      <button onClick={() => fetchNextPage()} disabled={!hasNextPage || isFetchingNextPage}>
        {isFetchingNextPage ? 'Đang tải...' : 'Tải thêm'}
      </button>
    </div>
  );
}

// GET https://jsonplaceholder.typicode.com/posts
// GET https://jsonplaceholder.typicode.com/posts/:id
// GET https://jsonplaceholder.typicode.com/users/:id

// Yêu cầu:
// 1. Danh sách bài viết (paginated, 10 bài/trang)
// 2. Chi tiết bài viết + thông tin tác giả
// 3. Bình luận của bài viết (lazy load khi click "Xem bình luận")
// 4. Loading skeleton (không phải spinner)
// 5. Error boundary + retry button
// 6. Search realtime (useQuery với searchTerm dependency)
// 7. Stale time: 5 phút
// 8. ReactQueryDevtools để xem cache

Câu hỏi ôn bài

1. queryKey trong TanStack Query có vai trò gì?

  • A. Chỉ là một cái tên để đặt cho query
  • B. Định danh duy nhất cho cache entry; khi queryKey thay đổi → fetch lại; cùng queryKey ở nhiều nơi → chia sẻ cache
  • C. Khóa API authentication
  • D. Thứ tự ưu tiên của các queries

Đáp án: B — queryKey là "địa chỉ" của cache entry. ['posts'] và ['posts', 1] là hai cache riêng biệt. Nhiều component dùng cùng queryKey → share cache → không duplicate request.

2. Sự khác biệt giữa isLoading và isFetching?

  • A. Chúng hoàn toàn giống nhau
  • B. isLoading: lần đầu, chưa có cache; isFetching: đang fetch kể cả background refetch
  • C. isFetching: chỉ khi user click refresh
  • D. isLoading: fetch từ network; isFetching: fetch từ cache

Đáp án: B — isLoading = isFetching && !data. Dùng isLoading để hiển thị skeleton lần đầu. Dùng isFetching để hiển thị spinner nhỏ khi background refetch.

3. Optimistic Update giải quyết vấn đề gì?

  • A. Tối ưu hóa code
  • B. Cập nhật UI ngay lập tức trước khi server phản hồi, tạo trải nghiệm nhanh hơn; rollback nếu thất bại
  • C. Tiết kiệm bandwidth
  • D. Giảm số lần re-render

Đáp án: B — Ví dụ: click Like → ngay lập tức UI hiện "đã like" mà không chờ server. Nếu API thành công → giữ nguyên. Nếu thất bại → rollback về "chưa like". UX mượt mà hơn nhiều.

4. Tại sao nên dùng queryClient.invalidateQueries sau mutation thành công?

  • A. Để xóa toàn bộ cache
  • B. Để đánh dấu cache liên quan là "stale" → tự động refetch → UI hiển thị data mới nhất từ server
  • C. Để ngăn người dùng mutation thêm lần nữa
  • D. Bắt buộc phải có nếu không ứng dụng crash

Đáp án: B — Sau khi POST/PUT/DELETE thành công, dữ liệu server đã thay đổi nhưng cache vẫn cũ. invalidateQueries đánh dấu stale → TanStack Query tự refetch → list cập nhật.

5. staleTime: 5 * 60 * 1000 có nghĩa là gì?

  • A. Cache hết hạn sau 5 phút, bị xóa khỏi bộ nhớ
  • B. Data được coi là "tươi" trong 5 phút — không refetch nếu navigate đi rồi quay lại trong 5 phút
  • C. Đợi 5 phút trước khi fetch lần đầu
  • D. Fetch lại mỗi 5 phút

Đáp án: B — staleTime: khoảng thời gian data được coi là "fresh". Trong thời gian này, navigate đi rồi về → dùng cache, không fetch lại. Sau staleTime, data "stale" → background refetch khi component mount.
